sudo apt install -y gpg curl
curl -s "https://apt.signet.org/keys/signet.asc" | \
    sudo gpg --dearmor -o /etc/apt/keyrings/signet.gpg
        
echo "deb [signed-by=/etc/apt/keyrings/signet.gpg] https://apt.signet.org/debian trixie main" \
    | sudo tee /etc/apt/sources.list.d/signet.list

apt update
apt install -y signet-gtk
